#!/bin/bash

# Update packages
yum update -y

# Install Apache HTTP Server
yum install httpd -y

# Start Apache
systemctl start httpd

# Enable Apache to start automatically after reboot
systemctl enable httpd

# Go to Apache web directory
cd /var/www/html

# Create the web page
echo "This is $(hostname)" > index.html

# Display the result
cat index.html

echo "Apache installation and configuration completed successfully."
------------------------------------------
# Make Executable 
chmod +x install-apache.sh

--------------------------
#Then Run it 
sudo ./install-apache.sh
